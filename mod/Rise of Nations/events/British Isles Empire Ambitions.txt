#Empire-ambition events for Britain's successors: unlike the border-claim events (one small
#contested zone each), these grant a sweeping batch of cores representing each nation's
#"empire of old" ambition, once it's an established power (great or secondary, top 16) under
#a reactionary government. add_core has no infamy cost, and neither does the acquire_all_cores
#CB it feeds (badboy_factor = 0), so this is meant to open a genuinely free, high-priority war
#path once it fires, not just flavor. One-shot per nation (set_country_flag on both options).
#IDs 99099025-99099027 continue from the claim events' range (99099017-99099024 used).

#ENL: the flagship "United Kingdom" ambition -- claims cores across every other successor's
#entire core territory via SCO/IRE/WHA/CRN = { all_core = { add_core = ENL } }, reusing each
#nation's own established core list rather than hardcoding province numbers.
country_event = {
    id = 99099025
    title = "EVTNAME99099025"
    desc = "EVTDESC99099025"

    trigger = {
        tag = ENL
        war = no
        OR = {
            is_greater_power = yes
            is_secondary_power = yes
        }
        ruling_party_ideology = reactionary
        NOT = { has_country_flag = united_kingdom_destiny_decided }
    }

    mean_time_to_happen = {
        months = 360
        modifier = {
            factor = 0.5
            war_policy = jingoism
        }
    }

    option = {
        name = "The Isles are ours by right"
        SCO = { all_core = { add_core = ENL } }
        IRE = { all_core = { add_core = ENL } }
        WHA = { all_core = { add_core = ENL } }
        CRN = { all_core = { add_core = ENL } }
        prestige = 10
        set_country_flag = united_kingdom_destiny_decided
    }
    option = {
        name = "Britain's division is Britain's peace"
        prestige = 2
        set_country_flag = united_kingdom_destiny_decided
    }
}
